function generateCoverLetter(){ const name = document.getElementById("clName").value.trim(); const job = document.getElementById("clJobTitle").value.trim(); const company = document.getElementById("clCompany").value.trim(); const skills = document.getElementById("clSkills").value.trim(); const description = document.getElementById("clDescription").value.trim(); const tone = document.getElementById("clTone").value; if(!name || !job || !company){ alert( "Please enter your name, job title and company." ); return; } let opening = "I am excited to apply for the " + job + " position at " + company + "."; if(tone === "confident"){ opening = "I am confident that my skills and experience make me a strong candidate for the " + job + " position at " + company + "."; } if(tone === "friendly"){ opening = "I would love the opportunity to join " + company + " as a " + job + "."; } if(tone === "concise"){ opening = "I am applying for the " + job + " position at " + company + "."; } const letter = ` Dear Hiring Manager, ${opening} My key skills include ${skills || "relevant technical and professional skills"}. I am motivated to contribute to your team, solve meaningful problems, and continue growing professionally. ${description ? "After reviewing the job requirements, I believe my background and skills align well with the responsibilities of this role." : "I believe my skills, motivation, and willingness to learn would allow me to contribute positively to your team." } Thank you for considering my application. I would welcome the opportunity to discuss how I can contribute to ${company}. Sincerely, ${name} `; document.getElementById( "coverLetterPreview" ).innerHTML = `

Cover Letter

${escapeHTML(letter).replace(/\n/g,"
")}
`; localStorage.setItem( "jobflow_cover_letter", letter ); } function copyCoverLetter(){ const text = localStorage.getItem("jobflow_cover_letter"); if(!text){ alert("Generate a cover letter first."); return; } if(navigator.clipboard){ navigator.clipboard.writeText(text) .then(() => { alert("Cover letter copied!"); }) .catch(() => { alert("Unable to copy. Please select and copy the text."); }); }else{ alert("Copy is not supported on this browser."); } } function createCoverLetterFromJob( title, company, description ){ const jobTitle = document.getElementById("clJobTitle"); const companyInput = document.getElementById("clCompany"); const descriptionInput = document.getElementById("clDescription"); if(jobTitle) jobTitle.value = title || ""; if(companyInput) companyInput.value = company || ""; if(descriptionInput) descriptionInput.value = description || ""; const name = localStorage.getItem("jobflow_name"); const nameInput = document.getElementById("clName"); if(nameInput && name) nameInput.value = name; if(typeof showPage === "function"){ showPage("coverLetter"); } } function downloadCoverLetter(){ const letter = localStorage.getItem("jobflow_cover_letter"); if(!letter){ alert("Generate a cover letter first."); return; } const blob = new Blob( [letter], { type: "text/plain;charset=utf-8" } ); const url = URL.createObjectURL(blob); const a = document.createElement("a"); a.href = url; a.download = "JobFlow-Cover-Letter.txt"; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); } function saveCoverLetter(){ const letter = localStorage.getItem("jobflow_cover_letter"); if(!letter){ alert("Generate a cover letter first."); return; } let saved = JSON.parse( localStorage.getItem("jobflow_saved_letters") || "[]" ); saved.unshift({ id: Date.now(), date: new Date().toLocaleDateString(), letter: letter }); localStorage.setItem( "jobflow_saved_letters", JSON.stringify(saved) ); alert("Cover letter saved successfully!"); } function loadSavedCoverLetters(){ const container = document.getElementById("savedCoverLetters"); if(!container) return; const saved = JSON.parse( localStorage.getItem( "jobflow_saved_letters" ) || "[]" ); if(saved.length === 0){ container.innerHTML = `
✉️

No saved cover letters

Generate and save your first cover letter to see it here.

`; return; } container.innerHTML = saved.map(item => { const preview = item.letter .replace(/\s+/g," ") .trim() .substring(0,180); return `

Cover Letter

Saved on ${escapeHTML(item.date)}
${escapeHTML(preview)}...
`; }).join(""); } function viewSavedCoverLetter(id){ const saved = JSON.parse( localStorage.getItem( "jobflow_saved_letters" ) || "[]" ); const item = saved.find( letter => letter.id === id ); if(!item){ alert( "Cover letter could not be found." ); return; } const preview = document.getElementById( "coverLetterPreview" ); if(!preview) return; preview.innerHTML = `

Saved Cover Letter

${escapeHTML(item.letter).replace(/\n/g,"
")}
`; } function copySavedCoverLetter(id){ const saved = JSON.parse( localStorage.getItem( "jobflow_saved_letters" ) || "[]" ); const item = saved.find( letter => letter.id === id ); if(!item) return; if(!navigator.clipboard){ alert( "Copy is not supported on this browser." ); return; } navigator.clipboard .writeText(item.letter) .then(() => { alert( "Cover letter copied!" ); }) .catch(() => { alert( "Unable to copy the cover letter." ); }); } function reuseCoverLetter(id){ const saved = JSON.parse( localStorage.getItem( "jobflow_saved_letters" ) || "[]" ); const item = saved.find( letter => letter.id === id ); if(!item){ alert( "Cover letter could not be found." ); return; } const nameInput = document.getElementById("clName"); const text = item.letter || ""; if(nameInput){ const lines = text .split("\n") .map(line => line.trim()) .filter(Boolean); const possibleName = lines.length > 0 ? lines[lines.length - 1] : ""; if(possibleName) nameInput.value = possibleName; } const descriptionInput = document.getElementById( "clDescription" ); if(descriptionInput){ descriptionInput.value = ""; } localStorage.setItem( "jobflow_cover_letter", item.letter ); if(typeof showPage === "function"){ showPage( "coverLetter" ); } const preview = document.getElementById( "coverLetterPreview" ); if(preview){ preview.innerHTML = `

Cover Letter

${escapeHTML(item.letter).replace(/\n/g,"
")}
`; } } function deleteSavedCoverLetter(id){ const confirmed = confirm( "Delete this saved cover letter?" ); if(!confirmed) return; let saved = JSON.parse( localStorage.getItem( "jobflow_saved_letters" ) || "[]" ); saved = saved.filter( letter => letter.id !== id ); localStorage.setItem( "jobflow_saved_letters", JSON.stringify(saved) ); loadSavedCoverLetters(); } function clearSavedCoverLetters(){ const saved = JSON.parse( localStorage.getItem( "jobflow_saved_letters" ) || "[]" ); if(saved.length === 0){ alert( "There are no saved cover letters." ); return; } const confirmed = confirm( "Delete all saved cover letters?" ); if(!confirmed) return; localStorage.removeItem( "jobflow_saved_letters" ); loadSavedCoverLetters(); } function searchSavedCoverLetters(){ const input = document.getElementById( "savedLetterSearch" ); const container = document.getElementById( "savedCoverLetters" ); if(!input || !container) return; const query = input.value .toLowerCase() .trim(); const saved = JSON.parse( localStorage.getItem( "jobflow_saved_letters" ) || "[]" ); const filtered = saved.filter(item => item.letter .toLowerCase() .includes(query) ); if(filtered.length === 0){ container.innerHTML = `

No matching letters

Try another search term.

`; return; } container.innerHTML = filtered.map(item => { const preview = item.letter .replace(/\s+/g," ") .trim() .substring(0,180); return `

Cover Letter

${escapeHTML(item.date)}

${escapeHTML(preview)}...

`; }).join(""); }
